begin_class Point {
    public integer x;
    public integer y;
    public void init(integer dx, integer dy) {
        x<-dx;
        y<-dy;    
    }
    public integer sum() {
        return x+y;    
    }
}
end_class

integer add(integer a,integer b) {
    integer r;
    r<-a+b;
    return r;
}

bool isPositive(integer v) {
    if v > 0 then 
        return true;
    endif;
    return false;
}
